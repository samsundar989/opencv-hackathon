```python
from riversight.vision import dino_features as D            # needs models/onnx/dinov2_small_224x392.onnx (MANIFEST)
from riversight.calib.scene_v2 import ShadowV2
net = D.load_net()                                           # once per cycle; ENGINE_* looked up by name (5.1 renames)
basis = dict(np.load("models/sites_v2/dino_pca_basis.npz"))
sh = ShadowV2("models/sites_v2/<cam>")                       # skip the site if the dir is missing
small = warp_small_safe(img, H, ref_shape)                   # the same small frame frame_features() uses
F = block_features(small).astype(np.float16).astype(np.float32)
P = D.project(D.tokens(net, small)[1:], basis).astype(np.float16)   # float16 = the training cache
raw = sh.predict_raw(F, P, frame.time)
ks2, x2, s2 = sh.kalman_step(KalmanState.from_json(state.get("kalman_v2")), frame.time.timestamp(), raw)
reading["shadow_stage_ft"] = x2; state["kalman_v2"] = ks2.to_json()
```
